// Product class for the Admin Panel
class Product {
  constructor(id, name, price, category) {
    this.id = id;
    this.name = name;
    this.price = price;
    this.category = category;
  }

  // Reduces the price by the given percentage (default 10%)
  applyDiscount(percent = 10) {
    if (percent <= 0 || percent > 100) {
      throw new RangeError("Discount must be between 1 and 100 percent");
    }
    const discountAmount = (this.price * percent) / 100;
    this.price = this.price - discountAmount;
    return this.price;
  }

  // Returns a formatted string with the product details
  getDetails() {
    return `ID: ${this.id} | ${this.name.padEnd(12)} | Category: ${this.category.padEnd(12)} | Price: Rs. ${this.price.toLocaleString("en-IN")}`;
  }
}

// Store multiple product objects in an array
const products = [
  new Product(1, "Laptop", 55000, "Electronics"),
  new Product(2, "Mouse", 499, "Accessories"),
  new Product(3, "Keyboard", 1500, "Accessories"),
  new Product(4, "Headphones", 2999, "Electronics"),
  new Product(5, "USB Cable", 199, "Accessories"),
  new Product(6, "Monitor", 12000, "Electronics")
];

console.log("===== ALL PRODUCTS =====");
products.forEach(product => console.log(product.getDetails()));

// Apply a 10% discount on one product and show the change
console.log("\n===== APPLY 10% DISCOUNT ON LAPTOP =====");
const laptop = products[0];
console.log("Before:", laptop.getDetails());
laptop.applyDiscount(10);
console.log("After :", laptop.getDetails());

// Filter products with price greater than 1000
console.log("\n===== PRODUCTS WITH PRICE > 1000 =====");
const expensiveProducts = products.filter(product => product.price > 1000);
expensiveProducts.forEach(product => console.log(product.getDetails()));
console.log(`Total: ${expensiveProducts.length} product(s)`);

    // Validation rules: one RegExp and one error message per field id
    const rules = {
      name: {
        regex: /^[A-Za-z]+(?: [A-Za-z]+)*$/,   // only alphabets (single spaces between words allowed)
        message: "Name must contain alphabets only."
      },
      email: {
        regex: /^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$/, // example@domain.com

      },
      phone: {
        regex: /^\d{10}$/,                      // exactly 10 digits
        message: "Phone must be exactly 10 digits."
      },
      password: {
        // lookaheads: 1 uppercase, 1 number, 1 special character (+ minimum 8 characters)
        regex: /^(?=.*[A-Z])(?=.*\d)(?=.*[^A-Za-z0-9]).{8,}$/,
        message: "Password needs 1 uppercase, 1 number, 1 special character (min 8 chars)."
      }
    };

    const form = document.getElementById("registrationForm");
    const inputs = form.querySelectorAll("input");
    const successMessage = document.getElementById("successMessage");

    // Validates one input: red border + message if invalid, green if valid
    function validateField(input) {
      const rule = rules[input.id];
      const errorBox = document.getElementById(`${input.id}Error`);
      const value = input.id === "password" ? input.value : input.value.trim();
      const isValid = rule.regex.test(value);

      input.classList.toggle("valid", isValid);
      input.classList.toggle("invalid", !isValid);
      errorBox.textContent = isValid ? "" : rule.message;
      return isValid;
    }

    // Live validation while typing
    inputs.forEach(input => input.addEventListener("input", () => validateField(input)));

    form.addEventListener("submit", event => {
      event.preventDefault();
      // map() first so that EVERY field gets validated (no short-circuit)
      const results = Array.from(inputs).map(validateField);
      const allValid = results.every(Boolean);
      successMessage.textContent = allValid ? "Registration successful!" : "";
    });
  </script>

// Book class with issue/return behaviour
class Book {
  constructor(title, author, isbn, isIssued = false) {
    this.title = title;
    this.author = author;
    this.isbn = isbn;
    this.isIssued = isIssued;
  }

  issueBook() {
    if (this.isIssued) {
      return false;               // already issued, cannot issue again
    }
    this.isIssued = true;
    return true;
  }

  returnBook() {
    if (!this.isIssued) {
      return false;               // was never issued
    }
    this.isIssued = false;
    return true;
  }
}

// Array of book objects
const library = [
  new Book("Clean Code", "Robert C. Martin", "978-0132350884"),
  new Book("Eloquent JavaScript", "Marijn Haverbeke", "978-1593279509"),
  new Book("You Don't Know JS", "Kyle Simpson", "978-1491904244"),
  new Book("The Pragmatic Programmer", "Andrew Hunt", "978-0135957059", true)
];

// Display all books that are NOT issued
function displayAvailableBooks() {
  console.log("----- Available Books -----");
  const available = library.filter(book => !book.isIssued);
  if (available.length === 0) {
    console.log("No books are available right now.");
    return;
  }
  available.forEach(book => {
    console.log(`${book.title} by ${book.author} (ISBN: ${book.isbn})`);
  });
}

// Search by ISBN and issue the book
function issueBookByISBN(isbn) {
  const book = library.find(item => item.isbn === isbn.trim());

  if (!book) {
    console.log(`[NOT FOUND] No book with ISBN ${isbn}`);
  } else if (book.issueBook()) {
    console.log(`[ISSUED] "${book.title}" has been issued.`);
  } else {
    console.log(`[UNAVAILABLE] "${book.title}" is already issued.`);
  }
}

function returnBookByISBN(isbn) {
  const book = library.find(item => item.isbn === isbn.trim());
  if (book && book.returnBook()) {
    console.log(`[RETURNED] "${book.title}" has been returned.`);
  } else {
    console.log(`[ERROR] Could not return book with ISBN ${isbn}`);
  }
}

// ---------- Demonstration ----------
displayAvailableBooks();

console.log("\n----- Issue books -----");
issueBookByISBN("978-0132350884"); // success
issueBookByISBN("978-0132350884"); // already issued
issueBookByISBN("978-0000000000"); // not found

console.log();
displayAvailableBooks();

console.log("\n----- Return a book -----");
returnBookByISBN("978-0132350884");
console.log();
displayAvailableBooks();

// Optional: let the user pass an ISBN from the command line
//   node q3.js 978-1593279509
const userIsbn = process.argv[2];
if (userIsbn) {
  console.log(`\n----- Issue from command line: ${userIsbn} -----`);
  issueBookByISBN(userIsbn);
  displayAvailableBooks();
}

  <div class="wrapper">

  </div>

  <script>
    // FormBuilder generates a form from an array of field objects
    class FormBuilder {
      constructor(containerId, fields, onSubmit) {
        this.container = document.getElementById(containerId);
        this.fields = fields;
        this.onSubmit = onSubmit;
        this.form = null;
      }

      // Escapes special characters so labels cannot break the generated HTML
      static escapeHTML(text) {
        const map = { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" };
        return String(text).replace(/[&<>"']/g, char => map[char]);
      }

      // Field name used as the key of the returned object (e.g. "Full Name" -> "full_name")
      getFieldName(field) {
        return field.name || field.label.toLowerCase().replace(/[^a-z0-9]+/g, "_").replace(/^_+|_+$/g, "");
      }

      // Builds the HTML string for one field
      buildField(field) {
        const name = FormBuilder.escapeHTML(this.getFieldName(field));
        const label = FormBuilder.escapeHTML(field.label);
        const required = field.required ? "required" : "";
        const allowedTypes = ["text", "email", "password", "number", "date", "tel"];

        if (field.type === "checkbox") {
          return `<div class="field"><label><input type="checkbox" id="${name}" name="${name}">${label}</label></div>`;
        }

        let control;
        if (field.type === "textarea") {
          control = `<textarea id="${name}" name="${name}" rows="3" ${required}></textarea>`;
        } else if (field.type === "select") {
          const options = (field.options || [])
            .map(option => `<option value="${FormBuilder.escapeHTML(option)}">${FormBuilder.escapeHTML(option)}</option>`)
            .join("");
          control = `<select id="${name}" name="${name}" ${required}>${options}</select>`;
        } else {
          const type = allowedTypes.includes(field.type) ? field.type : "text";
          control = `<input type="${type}" id="${name}" name="${name}" ${required}>`;
        }
        return `<div class="field"><label for="${name}">${label}</label>${control}</div>`;
      }

      // Creates the whole form dynamically using innerHTML
      render() {
        const fieldsHTML = this.fields.map(field => this.buildField(field)).join("");
        this.container.innerHTML = `<form id="dynamicForm">${fieldsHTML}<button type="submit">Submit</button></form>`;

        this.form = this.container.querySelector("#dynamicForm");
        this.form.addEventListener("submit", event => {
          event.preventDefault();               // stop page reload
          this.onSubmit(this.getFormData());    // pass collected values to the handler
        });
      }

      // Returns all entered values as an object
      getFormData() {
        const data = {};
        this.fields.forEach(field => {
          const name = this.getFieldName(field);
          const element = this.form.elements[name];
          data[name] = field.type === "checkbox" ? element.checked : element.value;
        });
        return data;
      }
    }

    // Field definitions
    const fields = [
      { type: "text", label: "Username", required: true },
      { type: "email", label: "Email", required: true },
      { type: "password", label: "Password", required: true },
      { type: "number", label: "Age" },
      { type: "select", label: "Course", options: ["B.Tech", "BCA", "MCA", "MBA"] },
      { type: "textarea", label: "About You" },
      { type: "checkbox", label: "I agree to the terms" }
    ];

    const output = document.getElementById("output");

    const builder = new FormBuilder("formContainer", fields, data => {
      output.textContent = JSON.stringify(data, null, 2); // show the object
      console.log("Form data:", data);
    });
    builder.render();
  </script>
</body>
</html>

// Assumption: "salary" is the MONTHLY salary, so annual = salary * 12
class Employee {
  constructor(id, name, department, salary) {
    this.id = id;
    this.name = name;
    this.department = department;
    this.salary = salary;
  }

  // Annual salary = monthly salary x 12
  getAnnualSalary() {
    return this.salary * 12;
  }

  // Increases the monthly salary by the given percentage
  applyBonus(percent) {
    if (percent <= 0) {
      throw new RangeError("Bonus percentage must be greater than 0");
    }
    this.salary += (this.salary * percent) / 100;
    return this.salary;
  }
}

// Create 5 employee objects
const employees = [
  new Employee(101, "Amit", "Engineering", 60000),
  new Employee(102, "Sara", "HR", 40000),
  new Employee(103, "Kiran", "Finance", 55000),
  new Employee(104, "Neha", "Marketing", 45000),
  new Employee(105, "Rahul", "Engineering", 70000)
];

// Formats numbers in Indian currency style
const formatMoney = amount => `Rs. ${amount.toLocaleString("en-IN")}`;

console.log("===== ANNUAL SALARY OF EACH EMPLOYEE =====");
employees.forEach(emp => {
  console.log(
    `${emp.id} | ${emp.name.padEnd(6)} | ${emp.department.padEnd(12)} | Monthly: ${formatMoney(emp.salary)} | Annual: ${formatMoney(emp.getAnnualSalary())}`
  );
});

// reduce() adds up every employee's annual salary
const totalAnnualPayout = employees.reduce(
  (total, emp) => total + emp.getAnnualSalary(),
  0
);
console.log(`\nTotal annual payout of the company: ${formatMoney(totalAnnualPayout)}`);

// Apply a 10% bonus to one employee and recalculate
console.log("\n===== APPLY 10% BONUS TO AMIT =====");
employees[0].applyBonus(10);
console.log(`Amit's new monthly salary: ${formatMoney(employees[0].salary)}`);
console.log(`Amit's new annual salary : ${formatMoney(employees[0].getAnnualSalary())}`);

const newTotal = employees.reduce((total, emp) => total + emp.getAnnualSalary(), 0);
console.log(`New total annual payout  : ${formatMoney(newTotal)}`);

  <script>
    // One RegExp per requirement so that we can show SPECIFIC error messages
    const usernameRules = [
      { regex: /^.{5,}$/, message: "Username must be at least 5 characters long." }
    ];

    const passwordRules = [
      { regex: /^.{8,}$/,       message: "Password must be at least 8 characters long." },
      { regex: /\d/,            message: "Password must include at least one number." },
      { regex: /[A-Z]/,         message: "Password must include at least one uppercase letter." },
      { regex: /[a-z]/,         message: "Password must include at least one lowercase letter." },
      { regex: /[^A-Za-z0-9]/,  message: "Password must include at least one special character." }
    ];

    // Returns the messages of all failed rules
    function getErrors(value, rules) {
      return rules.filter(rule => !rule.regex.test(value)).map(rule => rule.message);
    }

    const form = document.getElementById("loginForm");
    const messageBox = document.getElementById("messages");

    form.addEventListener("submit", event => {
      event.preventDefault();
      const username = document.getElementById("username").value.trim();
      const password = document.getElementById("password").value;

      const errors = [
        ...getErrors(username, usernameRules),

        });

class Cart {
  constructor() {
    this.items = [];
    this.discountPercent = 0;
    this.appliedCoupon = null;
  }

  // Adds an item with name, price and quantity
  addItem(name, price, quantity) {
    if (typeof name !== "string" || name.trim() === "") {
      throw new Error("Item name is required");
    }
    if (!(price > 0) || !Number.isInteger(quantity) || quantity < 1) {
      throw new Error(`Invalid price or quantity for "${name}"`);
    }
    this.items.push({ name: name.trim(), price, quantity });
  }

  // Total before any discount
  getTotal() {
    return this.items.reduce((total, item) => total + item.price * item.quantity, 0);
  }

  // Validates the coupon with a RegExp: SAVE20 or DISC10
  // Format: prefix SAVE or DISC followed by exactly two digits (the discount %)
  applyCoupon(code) {
    const couponPattern = /^(SAVE|DISC)(\d{2})$/;
    const match = couponPattern.exec(String(code).trim().toUpperCase());

    if (!match || Number(match[2]) === 0) {
      this.discountPercent = 0;
      this.appliedCoupon = null;
      return false; // invalid coupon
    }

    this.discountPercent = Number(match[2]); // e.g. "SAVE20" -> 20
    this.appliedCoupon = match[0];
    return true;
  }

  getDiscountAmount() {
    return (this.getTotal() * this.discountPercent) / 100;
  }

  getFinalTotal() {
    return this.getTotal() - this.getDiscountAmount();
  }

  // Prints the cart summary
  printSummary() {
    console.log("----- CART SUMMARY -----");
    this.items.forEach(item => {
      console.log(
        `${item.name.padEnd(12)} Rs. ${item.price} x ${item.quantity} = Rs. ${item.price * item.quantity}`
      );
    });
    console.log(`Subtotal      : Rs. ${this.getTotal().toFixed(2)}`);
    if (this.appliedCoupon) {
      console.log(`Coupon (${this.appliedCoupon}): -${this.discountPercent}% = -Rs. ${this.getDiscountAmount().toFixed(2)}`);
    }
    console.log(`Final Total   : Rs. ${this.getFinalTotal().toFixed(2)}`);
  }
}

// ---------- Demonstration ----------
const cart = new Cart();

  } else {
    console.log(`Coupon "${code}" is INVALID -> no discount applied`);
  }
});
