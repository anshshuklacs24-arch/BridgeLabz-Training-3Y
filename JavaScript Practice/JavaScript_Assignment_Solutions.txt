---------------------------------------------------------------------

// Variable for the user's name
const userName = "Ansh";

// Get the current hour (0-23) from the system clock
const currentHour = new Date().getHours();

// Variable to store the greeting message
let greeting;

if (currentHour < 12) {
  // Before 12 PM
  greeting = `Good Morning ${userName}!`;
} else if (currentHour >= 12 && currentHour <= 17) {
  // Between 12 PM and 5 PM
  greeting = `Good Afternoon ${userName}!`;
} else {
  // Otherwise (evening / night)
  greeting = `Good Evening ${userName}!`;
}

console.log(greeting);

FILE: q2.js  -  Multi-Type Data Summary
---------------------------------------------------------------------

// Variables of different data types
const userName = "Ansh Shukla";                       // string
const userAge = 21;                                   // number
const isStudent = true;                               // boolean
const subjects = ["JavaScript", "DBMS", "Networks"];  // array
const userProfile = { city: "Mathura", year: 3 };     // object
const middleName = null;                              // null
let phoneNumber;                                      // undefined

// Helper function: returns the correct type name for any value.
// typeof null gives "object" (a known JS quirk) and typeof [] also
// gives "object", so we handle null and arrays separately.
function getType(value) {
  if (value === null) return "null";
  if (Array.isArray(value)) return "array";
  return typeof value;
}

// Build the report rows: label + value + type
const report = [
  { label: "userName",    value: userName,    type: getType(userName) },
  { label: "userAge",     value: userAge,     type: getType(userAge) },
  { label: "isStudent",   value: isStudent,   type: getType(isStudent) },
  { label: "subjects",    value: JSON.stringify(subjects),    type: getType(subjects) },
  { label: "userProfile", value: JSON.stringify(userProfile), type: getType(userProfile) },
  { label: "middleName",  value: middleName,  type: getType(middleName) },
  { label: "phoneNumber", value: phoneNumber, type: getType(phoneNumber) }
];

// One single console.table() statement prints the whole report
console.table(report);

FILE: q3.js  -  Monthly Expense Tracker
---------------------------------------------------------------------

// Expenses for 5 categories: food, travel, rent, bills, leisure
const expenses = [4500, 2000, 12000, 3000, 1500];

// Add all expenses using a loop and the += assignment operator
let total = 0;
for (let i = 0; i < expenses.length; i++) {
  total += expenses[i];
}

// Average expense per category
const average = total / expenses.length;

// Add 10% tax to the total using arithmetic + assignment operators
let finalAmount = total;
finalAmount += total * 0.10;

// toFixed(2) rounds values to 2 decimal places
console.log("Total Expense      : " + total.toFixed(2));
console.log("Average Expense    : " + average.toFixed(2));
console.log("Final Amount (+10% tax): " + finalAmount.toFixed(2));

FILE: q4.js  -  Academic Performance Evaluator
---------------------------------------------------------------------

// Marks of 5 subjects (each out of 100)
const marks = [78, 85, 92, 66, 88];
const maxMarksPerSubject = 100;

// Calculate total marks
let totalMarks = 0;
for (let i = 0; i < marks.length; i++) {
  totalMarks += marks[i];
}

// Average marks and overall percentage
const average = totalMarks / marks.length;
const percentage = (totalMarks / (marks.length * maxMarksPerSubject)) * 100;

// Validation: check if any subject is below 35
let failedAnySubject = false;
for (let i = 0; i < marks.length; i++) {
  if (marks[i] < 35) {
    failedAnySubject = true;
    break;
  }
}

console.log("Average    : " + average.toFixed(2));
console.log("Percentage : " + percentage.toFixed(2) + "%");

// Decide result using logical operators
if (failedAnySubject || percentage < 50) {
  console.log("Detained");
} else if (percentage >= 85) {
  console.log("Promoted with Distinction");
} else if (percentage >= 50 && percentage < 85) {
  console.log("Promoted");
}

FILE: q9.js  -  Random Math Quiz Generator
---------------------------------------------------------------------

// Generate two random numbers between 1 and 20
const num1 = Math.floor(Math.random() * 20) + 1;
const num2 = Math.floor(Math.random() * 20) + 1;

// Pick a random operator from the array
const operators = ['+', '-', '*', '/'];
const operator = operators[Math.floor(Math.random() * operators.length)];

// Calculate the correct answer using a switch statement
let answer;

switch (operator) {
  case '+':
    answer = num1 + num2;
    break;
  case '-':
    answer = num1 - num2;
    break;
  case '*':
    answer = num1 * num2;
    break;
  case '/':
    // Division rounded to two decimals
    answer = (num1 / num2).toFixed(2);
    break;
  default:
    answer = "Invalid operator";
}

console.log(`Question: What is ${num1} ${operator} ${num2} ?`);
console.log(`Correct Answer: ${answer}`);
