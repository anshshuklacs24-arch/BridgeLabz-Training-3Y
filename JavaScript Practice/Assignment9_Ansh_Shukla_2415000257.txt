##  Q1.  WELCOME PAGE GREETING
##  FILES: q1.html  +  q1.js

PROBLEM
-------
Create a welcome page. 1) Greet the visitor on page load according to the
time of day (Morning / Afternoon / Evening). 2) A "Change Greeting" button
replaces the greeting with a motivational quote. 3) Another button shows or
hides a welcome message. 4) Clicking the greeting shows an alert.

ANALYSIS (what the question really asks)
----------------------------------------
* Hour of the day: before 12 = Morning, 12 to 16 = Afternoon, 17 onwards =
  Evening. The greeting is built inside $(function () {...}), so it runs when
  the DOM is ready.
* Extra: an optional name field adds ", <name>!" to the greeting while typing
  (keyup). Typing does not overwrite a quote that is on the screen.
* Change Greeting cycles through 5 quotes (fadeOut, new text, fadeIn). A Reset
  button brings the time-based greeting back.
* Toggle: .toggle() shows/hides the welcome message and the button text is
  updated ("Hide Welcome Message" / "Show Welcome Message").
* Clicking the greeting calls alert() with the current text.

CODE - FILE 2 of 2:  q1.js  (the jQuery code of this question)
------------------------------------------------------------
// Runs when the DOM is ready (all elements exist and can be selected)
$(document).ready(function () {

  // 5 motivational quotes, used by the "Change Greeting" button
  const quotes = [
    "Small steps every day lead to big results.",
    "Believe you can, and you are halfway there.",
    "Do something today that your future self will thank you for.",
    "Dream big, start small, act now.",
    "Every expert was once a beginner."
  ];
  let quoteIndex = 0;

  // Greeting based on the time of day: Morning / Afternoon / Evening
  function getTimeGreeting() {
    const hour = new Date().getHours();
    if (hour < 12) return "Good Morning";
    if (hour < 17) return "Good Afternoon";
    return "Good Evening";
  }

  // Personalised greeting: adds the visitor's name when one is typed
  function buildGreeting() {
    const name = $.trim($("#visitorName").val());   // $.trim() removes extra spaces
    return name ? getTimeGreeting() + ", " + name + "!" : getTimeGreeting() + "!";
  }

  // Scenario 1: greeting is shown on page load
  function showGreeting() {
    $("#greeting").stop(true, true).text(buildGreeting());
    $("#greeting").data("isQuote", false);          // .data() stores a flag on the element
  }
  showGreeting();

  // Typing a name updates the greeting (only while the time greeting is shown)
  $("#visitorName").on("keyup", function () {
    if (!$("#greeting").data("isQuote")) {
      showGreeting();
    }
  });

  // Scenario 2: "Change Greeting" replaces the text with a motivational quote
  $("#changeGreetingBtn").on("click", function () {
    const quote = quotes[quoteIndex];
    quoteIndex = (quoteIndex + 1) % quotes.length;  // next click shows the next quote
    $("#greeting")
      .data("isQuote", true)
      .stop(true, true)                             // finish any running animation first
      .fadeOut(150, function () {                   // fade out, change text, fade in
        $(this).text(quote).fadeIn(150);
      });
  });

  // "Reset Greeting" brings the time-based greeting back
  $("#resetGreetingBtn").on("click", showGreeting);

  // Scenario 3: toggle the visibility of the welcome message with another button
  $("#toggleWelcomeBtn").on("click", function () {
    const $message = $("#welcomeMessage").toggle(); // toggle() shows it if hidden, hides it if shown
    $(this).text($message.is(":visible") ? "Hide Welcome Message" : "Show Welcome Message");
  });

  // Scenario 4: clicking the greeting shows an alert
  $("#greeting").on("click", function () {
    alert('You clicked the greeting: "' + $(this).text() + '"');
  });
});

HOW TO TEST IT (open q1.html in the browser)
---------------------------------------------
1. On load the heading says Good Morning / Afternoon / Evening, as per your
   computer clock.
2. Type a name: it appears in the greeting. Click "Change Greeting" a few
   times: a different quote appears each time. Click Reset: greeting returns.
3. Click "Hide Welcome Message": the message disappears. Click again: it shows.
4. Click on the greeting text: an alert box shows what you clicked.

EXPLANATION
-----------
$(function () {...}) is the short form of $(document).ready(). .text() sets
the text safely (no HTML is parsed). .stop(true, true) before an animation
prevents a queue of animations when the button is clicked fast. .data() keeps
a small flag (isQuote) on the element, so the greeting knows what it shows.

PROBLEM
-------
Create a product list (at least 5 products). 1) Click a product to highlight
its background. 2) Hovering shows additional product details. 3) Clicking a
"Favorite" icon toggles a "selected" class. 4) Style discounted products
differently using an attribute selector. 5) Alert if a product is out of stock

ANALYSIS (what the question really asks)
----------------------------------------
* 6 products as <article class="product">. Discounted products carry
  data-discount="20" and out-of-stock ones data-stock="0".

  stock products (greyed, "Out of stock" label).
* Click: if data-stock is 0 an alert says so; otherwise the product gets the
  "highlighted" class (only one product is highlighted at a time).
* Hover: .hover(in, out) with slideDown / slideUp shows the details block.
* Favorite button: event.stopPropagation() is used so that clicking the heart
  does NOT also highlight the product. It toggles the "selected" class, changes
  the icon and updates the favourites counter.

CODE - FILE 2 of 2:  q2.js  (the jQuery code of this question)
------------------------------------------------------------
$(function () {
  const $products = $(".product");

  // ---------- Scenarios 4 and 5 use ATTRIBUTE SELECTORS ----------
  // [data-discount]  -> matches every product that HAS this attribute (discounted products)
  $(".product[data-discount]").each(function () {
    const percent = $(this).attr("data-discount");       // read the attribute value
    $(this)
      .addClass("has-discount")                           // different style for discounted products
      .find(".price")
      .after($("<span>", { "class": "badge", text: "-" + percent + "% OFF" }));
  });

  // [data-stock="0"] -> matches products whose stock attribute is exactly 0
  $('.product[data-stock="0"]')
    .addClass("out-of-stock")
    .append($("<p>", { "class": "stock-label", text: "Out of stock" }));

  // Scenario 2: hover shows additional details (hover = mouseenter + mouseleave)
  $products.hover(
    function () { $(this).find(".details").stop(true, true).slideDown(150); },
    function () { $(this).find(".details").stop(true, true).slideUp(150); }
  );

  // Scenarios 1 and 5: click highlights the product, or alerts if it is out of stock
  $products.on("click", function () {
    const $product = $(this);

    if (Number($product.attr("data-stock")) === 0) {
      alert('Sorry, "' + $product.attr("data-name") + '" is out of stock.');
      return;                                             // out-of-stock products are not highlighted
    }

    const wasHighlighted = $product.hasClass("highlighted");
    $products.removeClass("highlighted");                 // only one product is highlighted at a time
    if (!wasHighlighted) {
      $product.addClass("highlighted");                   // clicking the same product again removes it
    }
  });

  // Scenario 3: the Favourite icon toggles the "selected" class
  $(".fav").on("click", function (event) {
    event.stopPropagation();                              // do not trigger the product's own click
    const $icon = $(this).toggleClass("selected");        // adds the class if missing, removes it if present
    const isSelected = $icon.hasClass("selected");
    $icon.html(isSelected ? "&#9829;" : "&#9825;").attr("aria-pressed", String(isSelected));
    $("#favCount").text("Favourites: " + $(".fav.selected").length);
  });
});

HOW TO TEST IT (open q2.html in the browser)
---------------------------------------------
1. Click a product: its background turns yellow. Click another: it moves.
2. Move the mouse over a product: extra details slide open; leave: they close.
3. Click the heart on a product: it turns red/filled (class "selected"). The
   product itself is not highlighted by this click. Click again: it resets.
4. Headphones, Backpack and Gaming Mouse show a discount badge and colour.
5. Click "Smart Watch" or "Gaming Mouse": an out of stock alert appears.

EXPLANATION
-----------
stopPropagation() is the key detail: the heart is INSIDE the product, so a
click on it would bubble up to the product's own click handler. Attribute
selectors keep the logic in the HTML data, not in hard-coded product names.

PROBLEM
-------
Create an FAQ with at least 5 questions. 1) Clicking a question toggles its
answer. 2) Hover changes the question colour. 3) Double-click on a question
collapses ALL answers. 4) Focus on an answer input highlights the parent
question. 5) Blur resets the background colour.

ANALYSIS (what the question really asks)
----------------------------------------
* 5 .faq-item articles: h3.question, div.answer (hidden by CSS) and an
  input.answer-input inside the answer.
* Click: $(this).next(".answer").slideToggle() - the answer is the next
  sibling of the question.
* Hover: .hover() sets a new text colour and removes it again.
* Double-click: $(".answer").slideUp() closes every answer.
* Focus / blur on the input: .closest(".faq-item").find(".question") reaches
  the parent question; .css("background-color", ...) highlights it on focus
  and resets it (empty string) on blur.
* Accessibility: questions have tabindex="0", so Enter or Space also toggles.

$(function () {

  // Returns the banner chosen in the drop-down: all banners, or one banner by its data-id
  function getTargetBanners() {
    const choice = $("#bannerTarget").val();
    return choice === "all" ? $(".banner") : $('.banner[data-id="' + choice + '"]');
  }

  // Scenario 1: Hide  -> hides immediately (no animation)
  $("#hideBtn").on("click", function () { getTargetBanners().hide(); });

  // Scenario 2: Show  -> shows hidden banners again
  $("#showBtn").on("click", function () { getTargetBanners().show(); });

  // Scenario 3: Slide Up / Slide Down -> the height animates to 0 and back
  $("#slideUpBtn").on("click", function () { getTargetBanners().slideUp(400); });
  $("#slideDownBtn").on("click", function () { getTargetBanners().slideDown(400); });

  // Scenario 4: Fade Out / Fade In -> opacity animates, so banners vanish and appear gradually
  $("#fadeOutBtn").on("click", function () { getTargetBanners().fadeOut(600); });
  $("#fadeInBtn").on("click", function () { getTargetBanners().fadeIn(600); });

  // ---------- Scenario 5: automatic rotation every 5 seconds ----------
  const $slides = $(".slide");
  let currentSlide = 0;
  let rotationTimer = null;

  function showNextSlide() {
    const $oldSlide = $slides.eq(currentSlide);
    currentSlide = (currentSlide + 1) % $slides.length;        // after the last slide, go back to the first

    // fade the old slide out, and in its callback fade the next one in
    $oldSlide.fadeOut(400, function () {
      $slides.eq(currentSlide).fadeIn(400);
      $("#slideIndicator").text("Offer " + (currentSlide + 1) + " of " + $slides.length);
    });
  }

  function startRotation() {
    if (rotationTimer === null) {
      rotationTimer = setInterval(showNextSlide, 5000);         // 5000 ms = 5 seconds
    }
  }

  function stopRotation() {
    clearInterval(rotationTimer);
    rotationTimer = null;
  }

  // The rotation pauses while the mouse is over the offer, so it can be read calmly
  $("#rotator").on("mouseenter", stopRotation).on("mouseleave", startRotation);

  startRotation();
});

$(function () {

  // Scenario 1: click a manager -> highlight all DIRECT reports
  // A report stores its manager's id in data-manager, so an attribute selector finds them.
  $(".employee.manager").on("click", function () {
    const managerId = $(this).attr("data-id");
    $(".employee").removeClass("report-highlight selected sibling-highlight");

    const $reports = $('.employee[data-manager="' + managerId + '"]').addClass("report-highlight");
    $("#status").text($.trim($(this).find(".emp-name").clone().children().remove().end().text()) +
      " has " + $reports.length + " direct reports");
  });

  // Scenario 2: hover an employee -> show contact info
  // .next() returns the element directly AFTER the name paragraph, which is the contact paragraph.
  $(".employee").hover(
    function () { $(this).children(".emp-name").next(".contact").stop(true, true).slideDown(150); },
    function () { $(this).children(".emp-name").next(".contact").stop(true, true).slideUp(150); }
  );

  // Scenario 3: click a department name -> change the background of ALL members of that department
  // .siblings(".members") finds the list next to the heading, .children() returns its <li> members.
  $(".dept-name").on("click", function () {
    const $members = $(this).siblings(".members");
    const color = $(this).closest(".department").attr("data-color");

    if ($members.data("colored")) {                       // second click removes the colour again
      $members.children().css("background-color", "");
      $members.data("colored", false);
    } else {
      $members.children().css("background-color", color);
      $members.data("colored", true);
    }
  });

  // Scenario 4: select a random employee -> highlight his/her siblings (same department list)
  $("#randomBtn").on("click", function () {
    const $all = $(".employee");
    $all.removeClass("report-highlight selected sibling-highlight");

    const randomIndex = Math.floor(Math.random() * $all.length);
    const $picked = $all.eq(randomIndex).addClass("selected");
    const $siblings = $picked.siblings().addClass("sibling-highlight");   // .siblings() = the other <li> in the same <ul>

    $("#status").text("Random pick: " + $.trim($picked.find(".emp-name").clone().children().remove().end().text()) +
      " (" + $siblings.length + " siblings highlighted)");
  });

  // Scenario 5: collapse / expand a team
  // .parent() goes from the button up to its department, .find(".members") goes down to the list.
  $(".toggle-team").on("click", function () {
    const $button = $(this);
    const $team = $button.parent().find(".members");

    $team.stop(true, true).slideToggle(250, function () {
      const isOpen = $team.is(":visible");
      $button.text(isOpen ? "Collapse team" : "Expand team").attr("aria-expanded", String(isOpen));
    });
  });
});

HOW TO TEST IT (open q5.html in the browser)
---------------------------------------------
1. Click the manager "Sara" in Engineering: her 4 reports are highlighted.
2. Hover any employee: the contact line appears below the name.
3. Click a department heading: every member of it gets the department colour.
4. Click "Select Random Employee": one employee is marked, its siblings in
   the same team are highlighted, and the status line names them.
5. Click "Collapse team": the list hides and the button says "Expand team".

EXPLANATION
-----------
Traversal methods: .next() = next sibling, .siblings() = all others at the
same level, .children() = direct children only, .parent() = one level up,
.find() = any depth down, .closest() = nearest ancestor. Using data attributes
(data-manager) makes the relationship readable and easy to extend.

$(function () {
  const $topicList = $("#topicList");

  // Scenario 5: a success message is CREATED with jQuery and inserted into the DOM, then removed
  function showMessage(text) {
    const $message = $("<p>", { "class": "success-message", text: text });
    $("#messages").append($message);                       // dynamically inserted into the DOM
    $message.hide().fadeIn(200).delay(2500).fadeOut(400, function () {
      $(this).remove();                                    // removed from the DOM after it faded out
    });
  }

  // Shows how many topics currently have notifications enabled
  function updateSummary() {
    $("#summary").text("Active notifications: " + $(".topic.subscribed").length);
  }

  // ---------- Event handlers (named functions, so .on() and .off() can use them) ----------

  // Scenario 1: Subscribe -> enable notifications for this topic
  function onSubscribe() {
    const $topic = $(this).closest(".topic");
    $topic.addClass("subscribed");
    $topic.find(".status").text("Notifications ON");
    $topic.find(".subscribe").prop("disabled", true);      // .prop() changes the "disabled" property
    $topic.find(".unsubscribe").prop("disabled", false);
    updateSummary();
    showMessage("Subscribed to " + $topic.attr("data-topic") + ". Notifications enabled.");
  }

  // Scenario 2: Unsubscribe -> disable notifications for this topic
  function onUnsubscribe() {
    const $topic = $(this).closest(".topic");
    $topic.removeClass("subscribed");
    $topic.find(".status").text("Notifications OFF");
    $topic.find(".subscribe").prop("disabled", false);
    $topic.find(".unsubscribe").prop("disabled", true);
    updateSummary();
    showMessage("Unsubscribed from " + $topic.attr("data-topic") + ". Notifications disabled.");
  }

  // Scenario 4: Remove a subscription -> detach its events with .off(), then delete the item
  function onRemove() {
    const $topic = $(this).closest(".topic");
    const topicName = $topic.attr("data-topic");

    // .off("click") removes every click handler that .on("click", ...) attached to these buttons
    $topic.find(".subscribe, .unsubscribe, .remove").off("click");

    $topic.fadeOut(300, function () {
      $(this).remove();
      updateSummary();
    });
    showMessage('Removed "' + topicName + '". Its click events were detached with .off().');
  }

  // Attaches the three click events to the buttons of ONE topic item with .on()
  function bindTopicEvents($topic) {
    $topic.find(".subscribe").on("click", onSubscribe);
    $topic.find(".unsubscribe").on("click", onUnsubscribe);
    $topic.find(".remove").on("click", onRemove);
  }

  // Attach events to the 5 topics that are already in the page
  $(".topic").each(function () {
    bindTopicEvents($(this));
  });

  // Builds the <li> of a new topic (same structure as the topics in the HTML)
  function createTopicItem(name) {
    return $("<li>", { "class": "topic", "data-topic": name }).append(
      $("<span>", { "class": "topic-name", text: name }),
      $("<span>", { "class": "status", text: "Notifications OFF" }),
      $("<span>", { "class": "actions" }).append(
        $("<button>", { type: "button", "class": "subscribe", text: "Subscribe" }),
        $("<button>", { type: "button", "class": "unsubscribe", text: "Unsubscribe", disabled: true }),
        $("<button>", { type: "button", "class": "remove", text: "Remove" })
      )
    );
  }

  // Scenario 3: dynamically add a new topic and attach .on() click events to its buttons
  $("#addTopicForm").on("submit", function (event) {
    event.preventDefault();
    const name = $.trim($("#newTopic").val());

    if (name === "") {
      showMessage("Please type a topic name first.");
      return;
    }
    // reject a topic that exists already (case-insensitive)
    const exists = $(".topic").filter(function () {
      return $(this).attr("data-topic").toLowerCase() === name.toLowerCase();
    }).length > 0;
    if (exists) {
      showMessage('"' + name + '" is already in the list.');
      return;
    }

    const $newTopic = createTopicItem(name);
    $newTopic.hide().appendTo($topicList).fadeIn(300);
    bindTopicEvents($newTopic);                             // new elements need their events attached
    $("#newTopic").val("").focus();
    showMessage('Topic "' + name + '" added.');
  });

  updateSummary();
});

CODE - FILE 2 of 2:  q7.js  (the jQuery code of this question)
------------------------------------------------------------
$(function () {
  const $courses = $(".course");
  const $texts = $courses.find(".course-title, .course-desc");

  // Remember the ORIGINAL text of every title/description. Highlighting changes the
  // inner HTML, so every search starts again from this clean text.
  $texts.each(function () {
    $(this).data("original", $(this).text());
  });

  // Escapes characters that have a special meaning in a regular expression (. * + ? ( ) etc.)
  function escapeRegExp(text) {
    return text.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
  }

  // Writes the text into the element and wraps every match in <span class="match">
  function highlightText($element, original, query) {
    $element.empty();
    if (query === "") {
      $element.text(original);
      return;
    }
    // split() with a capturing group keeps the matches: odd positions = matched parts
    const parts = original.split(new RegExp("(" + escapeRegExp(query) + ")", "gi"));
    $.each(parts, function (index, part) {
      if (index % 2 === 1) {
        $("<span>", { "class": "match", text: part }).appendTo($element);
      } else {
        $element.append(document.createTextNode(part));    // text nodes: no HTML injection possible
      }
    });
  }

  // Scenarios 1, 2, 3, 4: filter, highlight, toggle visibility, show the count
  function filterCourses() {
    const query = $.trim($("#searchInput").val());
    const lowerQuery = query.toLowerCase();
    let matchedCount = 0;

    $courses.each(function () {
      const $course = $(this);
      let isMatch = (lowerQuery === "");                    // an empty search matches every course

      $course.find(".course-title, .course-desc").each(function () {
        const original = $(this).data("original");
        if (lowerQuery !== "" && original.toLowerCase().indexOf(lowerQuery) !== -1) {
          isMatch = true;
        }
        highlightText($(this), original, query);
      });

      // Scenario 3: toggle(true) shows the course, toggle(false) hides it
      $course.toggle(isMatch);
      if (isMatch) matchedCount++;
    });

    // Scenario 2: highlight the matched text with .css()
    $(".match").css({ "background-color": "#ffe066", "font-weight": "bold", "padding": "0 2px" });

    // Scenario 4: show the number of matched courses dynamically
    $("#resultCount").text("Showing " + matchedCount + " of " + $courses.length + " courses");
    $("#noResults").toggle(matchedCount === 0);
  }

  // Scenario 1: filter in real time on every key release (.keyup()).
  // The "input" event is added too, so pasting with the mouse also works.
  $("#searchInput").keyup(filterCourses).on("input", filterCourses);

  // Scenario 5: Clear -> empty the box and reset the list to show all courses
  $("#clearBtn").on("click", function () {
    $("#searchInput").val("").focus();
    filterCourses();
  });

  filterCourses();
});

3. Type "c++(": nothing breaks (regex characters are escaped).

EXPLANATION

The search is case-insensitive (the "i" flag and toLowerCase()). Splitting

are the matches. That is why index % 2 === 1 decides which part to wrap.

PROBLEM
-------
A page uses TWO versions of jQuery. 1) Version 1 handles the carousel
slider rotation. 2) Version 2 manages modal popups for notifications.
3) Version 1 highlights the active widget. 4) Version 2 attaches tooltips on
hover. 5) Use jQuery.noConflict() so both versions operate simultaneously.

ANALYSIS (what the question really asks)
----------------------------------------
* Why is this a problem? Every jQuery file writes itself into the global
  names window.$ and window.jQuery. If two versions are loaded, the one
  loaded LAST overwrites the first.
* Loading order: jQuery 1.10.2 (Version 1) first, then jQuery 3.6.0
  (Version 2), then q9.js.
* The solution (first lines of q9.js):

      var $v1 = window.jQuery;             // Version 1 = 1.10.2
  noConflict(true) is called on the LAST loaded copy (3.6.0). It returns that
  copy (stored in $v2) and gives $ and jQuery back to the copy loaded before
  it (1.10.2, stored in $v1). Now both versions live in their own variables.
* Each version's code is wrapped in an IIFE that passes its jQuery in as "$":
  (function ($) { ... })($v1);   and   (function ($) { ... })($v2);
  Inside, the familiar $ is used and always means the correct version.
* Version 1: carousel (auto-rotates every 3 seconds with fade, Previous /
  Next buttons, pauses on hover) and the active-widget highlight.
* Version 2: notification modal (opens with fadeIn; closes by the button, by a
  click on the dark background, or the Esc key) and tooltips from the

// ======================= jQuery.noConflict() =======================
// Both jQuery files were loaded: 1.10.2 first, then 3.6.0. Every jQuery file
// writes itself into the global names  window.$  and  window.jQuery,
// so the file loaded LAST (3.6.0) has replaced the first one.
//
// jQuery.noConflict(true) is called on the LAST loaded copy:
//   1. it RETURNS that copy (3.6.0)  -> we store it in $v2
//   2. it gives window.$ and window.jQuery BACK to the previously loaded copy (1.10.2)
// Result: two independent jQuery objects that can be used at the same time.
var $v2 = jQuery.noConflict(true);   // Version 2 = 3.6.0
var $v1 = window.jQuery;             // Version 1 = 1.10.2 (restored by noConflict)

// Show which version is which (.fn.jquery contains the version number)
$v1("#versionInfo").text("Version 1 = jQuery " + $v1.fn.jquery + "   |   Version 2 = jQuery " + $v2.fn.jquery);

(function ($) {
  $(document).ready(function () {

    // Scenario 1: carousel slider rotation (fadeOut / fadeIn every 3 seconds)
    const $slides = $("#slides .slide");
    let current = 0;
    let timer = null;

    function showSlide(newIndex) {
      const $old = $slides.eq(current);
      current = (newIndex + $slides.length) % $slides.length;     // wraps around at both ends
      $old.stop(true, true).fadeOut(300, function () {
        $slides.eq(current).fadeIn(300);
        $("#slideInfo").text("Slide " + (current + 1) + " of " + $slides.length);
      });
    }

    function startCarousel() {
      if (timer === null) {
        timer = setInterval(function () { showSlide(current + 1); }, 3000);
      }
    }
    function stopCarousel() {
      clearInterval(timer);
      timer = null;
    }

    $("#nextSlide").on("click", function () { showSlide(current + 1); });
    $("#prevSlide").on("click", function () { showSlide(current - 1); });
    $("#slides").on("mouseenter", stopCarousel).on("mouseleave", startCarousel);   // pause on hover
    startCarousel();

    // Scenario 3: highlight the active widget (the one that was clicked last)
    $(".widget").on("click", function () {
      const wasActive = $(this).hasClass("active-widget");
      $(".widget").removeClass("active-widget");
      if (!wasActive) {
        $(this).addClass("active-widget");

})($v1);

$(function () {
  const $form = $("#registrationForm");
  const $fields = $form.find("input");

  // Emails that are already registered (5 sample accounts). New registrations are added here too.
  const registeredEmails = [
    "ansh@example.com", "riya@example.com", "karan@example.com",
    "meera@example.com", "rahul@example.com"
  ];
  const emailPattern = /^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$/;

  // One validator per field: returns an error message, or "" when the value is valid
  const validators = {
    // Scenario 1: name must not be empty
    name: function (value) {
      return value === "" ? "Name is required." : "";
    },
    // Scenario 2: email must have a valid format AND must be unique
    email: function (value) {
      if (value === "") return "Email is required.";
      if (!emailPattern.test(value)) return "Enter a valid email (name@example.com).";
      if ($.inArray(value.toLowerCase(), registeredEmails) !== -1) return "This email is already registered.";
      return "";
    },
    // Scenario 3: password must have at least 8 characters
    password: function (value) {
      return value.length < 8 ? "Password must be at least 8 characters." : "";
    }
  };

  // Checks one input, shows its message and colours its border. Returns true when valid.
  function validateField($input) {
    const fieldName = $input.attr("name");
    // passwords are not trimmed, the other values are
    const value = fieldName === "password" ? $input.val() : $.trim($input.val());
    const message = validators[fieldName](value);

    $input.siblings(".error").text(message);

    // Scenario 5: red border for an invalid field with .css(), green for a valid one
    $input.css("border-color", message === "" ? "#2e9e5b" : "red");
    return message === "";
  }

  // Live feedback: a field is re-checked while typing after it was touched once
  $fields.on("blur", function () {
    $(this).data("touched", true);
    validateField($(this));
  });
  $fields.on("keyup", function () {
    $("#successMessage").hide();                            // the old success message is outdated
    if ($(this).data("touched")) {
      validateField($(this));
    }
  });

  $form.on("submit", function (event) {
    event.preventDefault();                                 // stop the real form submission

    // map() validates EVERY field (no early exit), so all errors are shown at once
    const results = $fields.map(function () {
      $(this).data("touched", true);
      return validateField($(this));
    }).get();                                               // .get() turns the jQuery result into a plain array

    const isFormValid = $.inArray(false, results) === -1;

    if (isFormValid) {
      // Scenario 4: success message
      registeredEmails.push($.trim($("#email").val()).toLowerCase());   // this email is taken from now on
      $form[0].reset();
      $fields.css("border-color", "").data("touched", false);
      $form.find(".error").text("");
      $("#successMessage").fadeIn(300);
    } else {
      $("#successMessage").hide();
      $fields.filter(function () {                          // focus the first invalid field
        return $(this).css("border-color") === "rgb(255, 0, 0)";
      }).first().focus();
    }
  });
});

-----------
The uniqueness check uses $.inArray(), which returns -1 when the value is
not found. A real website must also check uniqueness on the SERVER, because
