##  Q1.  DYNAMIC PRODUCT LIST MANAGER (Add, Edit, Delete - Event Delegation)
##  FILES: q1.html  +  q1.js
##########################################################################

PROBLEM
-------
Create a product list application where:
 * User can add new products using an input box
 * Each product appears with Edit and Delete buttons
 * Editing allows inline text editing
 * Use event delegation to handle Edit/Delete on parent <ul>
 * If user clicks outside the edited item, the edit mode should auto-save

ANALYSIS (what the question really asks)
----------------------------------------
* The list is built at run time, so items do not exist when the page loads.
  A listener on every button would have to be added for each new item.
  EVENT DELEGATION avoids this: ONE click listener on the parent <ul> checks
  event.target.closest(".edit-btn") / ".delete-btn" and acts on the right item.
* Inline editing: when Edit is clicked, the name <span> is replaced by a text
  <input> showing the current name. The button changes to "Save".
* Auto-save: a click listener on the document saves the item being edited
  whenever the click target is NOT inside that item.
* Extra rules for a robust page: Enter saves, Esc cancels, empty text keeps
  the old name, blank product names are not added, and only one item is
  edited at a time (starting a second edit saves the first one).
* textContent is used (not innerHTML), so typed text can never inject HTML.

CODE - FILE 2 of 2:  q1.js  (the JavaScript of this question)
------------------------------------------------------------
"use strict";

const productInput = document.getElementById("productInput");
const addButton = document.getElementById("addButton");
const productList = document.getElementById("productList");
const emptyMessage = document.getElementById("emptyMessage");

// The <li> that is currently in edit mode (only one item can be edited at a time)
let editingItem = null;

// Shows the "No products yet" message only when the list is empty
function updateEmptyMessage() {
  emptyMessage.hidden = productList.children.length > 0;
}

// Builds one <li>: product name + Edit button + Delete button
function createProductItem(productName) {
  const item = document.createElement("li");
  item.className = "product-item";

  const nameSpan = document.createElement("span");
  nameSpan.className = "product-name";
  nameSpan.textContent = productName; // textContent is safe (no HTML injection)

  const editButton = document.createElement("button");
  editButton.type = "button";
  editButton.className = "edit-btn";
  editButton.textContent = "Edit";

  const deleteButton = document.createElement("button");
  deleteButton.type = "button";
  deleteButton.className = "delete-btn";
  deleteButton.textContent = "Delete";

  item.append(nameSpan, editButton, deleteButton);
  return item;
}

function addProduct() {
  const productName = productInput.value.trim();
  if (productName === "") {
    productInput.focus(); // ignore empty names
    return;
  }
  productList.appendChild(createProductItem(productName));
  productInput.value = "";
  productInput.focus();
  updateEmptyMessage();
}

// Switches an item to edit mode: the name <span> is replaced by a text <input>
function startEdit(item) {
  if (editingItem === item) return;
  if (editingItem) saveEdit(editingItem); // finish the previous edit first

  const nameSpan = item.querySelector(".product-name");
  const editInput = document.createElement("input");
  editInput.type = "text";
  editInput.className = "edit-input";
  editInput.value = nameSpan.textContent;
  editInput.dataset.original = nameSpan.textContent; // remembered for Esc / empty text

  nameSpan.replaceWith(editInput);
  item.querySelector(".edit-btn").textContent = "Save";
  item.classList.add("editing");
  editingItem = item;

  editInput.focus();
  editInput.select();
}

// Leaves edit mode and keeps the new text (empty text restores the old name)
function finishEdit(item, keepChanges) {
  const editInput = item.querySelector(".edit-input");
  if (!editInput) return;

  const newName = keepChanges ? editInput.value.trim() : "";
  const nameSpan = document.createElement("span");
  nameSpan.className = "product-name";
  nameSpan.textContent = newName || editInput.dataset.original;

  editInput.replaceWith(nameSpan);
  item.querySelector(".edit-btn").textContent = "Edit";
  item.classList.remove("editing");
  if (editingItem === item) editingItem = null;
}

const saveEdit = item => finishEdit(item, true);
const cancelEdit = item => finishEdit(item, false);

// ---------- EVENT DELEGATION ----------
// ONE listener on the parent <ul> handles the buttons of ALL items, including
// items that are added later. event.target tells us which button was clicked.
productList.addEventListener("click", event => {
  const item = event.target.closest(".product-item");
  if (!item) return;

  if (event.target.closest(".delete-btn")) {
    if (editingItem === item) editingItem = null;
    item.remove();
    updateEmptyMessage();
  } else if (event.target.closest(".edit-btn")) {
    if (editingItem === item) {
      saveEdit(item); // the button shows "Save" while editing
    } else {
      startEdit(item);
    }
  }
});

// Delegated keyboard handling inside the edit box: Enter = save, Esc = cancel
productList.addEventListener("keydown", event => {
  if (!event.target.classList.contains("edit-input")) return;
  const item = event.target.closest(".product-item");
  if (event.key === "Enter") {
    event.preventDefault();
    saveEdit(item);
  } else if (event.key === "Escape") {
    cancelEdit(item);
  }
});

// ---------- AUTO-SAVE ----------
// A click anywhere outside the item that is being edited saves it.
document.addEventListener("click", event => {
  if (editingItem && !editingItem.contains(event.target)) {
    saveEdit(editingItem);
  }
});

addButton.addEventListener("click", addProduct);
productInput.addEventListener("keydown", event => {
  if (event.key === "Enter") addProduct();
});

updateEmptyMessage();

HOW TO TEST IT (open q1.html in the browser)
---------------------------------------------
1. Type "Laptop", click Add. Add "Mouse" and "Phone" (Enter also adds).
2. Click Edit on "Mouse": a text box appears and the button says Save.
3. Change the text to "Gaming Mouse" and click on an empty part of the page:
   the new name is saved automatically (auto-save).
4. Click Edit, type, press Enter -> saved. Press Esc instead -> cancelled.
5. Click Edit on one item and then Edit on another: the first is saved.
6. Click Delete -> the item is removed. With no items, a message appears.

EXPLANATION
-----------
Why delegation works: a click on a button "bubbles" up to its parents, so
the <ul> receives it even though the button was created later. closest()
finds the <li> that owns the clicked button.

Order of handlers for a click on a second item's Edit button: the <ul>
listener runs first (saves the old item, starts the new edit), then the
document listener runs and sees that the click is inside the item being
edited, so it does not save again.

##  Q2.  LIVE CHARACTER COUNTER WITH WARNING BEHAVIOR
##  FILES: q2.html  +  q2.js
##########################################################################

PROBLEM
-------
Build a text area that:
 * Shows remaining characters (max 100 char)
 * Turns the counter yellow at 20 chars left
 * Turns red at 0 and prevents further typing using preventDefault()
 * A button "Reset" clears everything

ANALYSIS (what the question really asks)
----------------------------------------
* remaining = 100 - textarea.value.length, updated on every change.
* Colours are CSS classes toggled from JavaScript: "warning" (yellow) when
  remaining is 20 or less (and above 0), "danger" (red) at 0.
* Blocking: the maxlength attribute is NOT used, because the question wants
  preventDefault(). A keydown listener cancels any key that would add a
  character once the length is 100. Keys that add nothing (Backspace, arrows,
  Ctrl+C) and typing over a selected text are still allowed.
* Pasting could bypass the keydown rule, so a paste listener inserts only as
  many characters as still fit. An input listener is a safety net for
  drag-and-drop and phone keyboards.
* Reset empties the box, restores "100 characters left" and removes the
  colours.

CODE - FILE 2 of 2:  q2.js  (the JavaScript of this question)
------------------------------------------------------------
"use strict";

const MAX_CHARACTERS = 100;
const WARNING_LIMIT = 20; // counter turns yellow at 20 characters left

const textBox = document.getElementById("textBox");
const counter = document.getElementById("counter");
const resetButton = document.getElementById("resetButton");

// Updates the number and the colour of the counter
function updateCounter() {
  const remaining = MAX_CHARACTERS - textBox.value.length;
  counter.textContent = `${remaining} ${remaining === 1 ? "character" : "characters"} left`;

  counter.classList.toggle("danger", remaining <= 0);                           // red at 0
  counter.classList.toggle("warning", remaining > 0 && remaining <= WARNING_LIMIT); // yellow at 20 or less
}

// Keys that never add a character (Backspace, arrows, Ctrl+C ...) must stay allowed.
// A key adds a character if key.length === 1 (a, 5, space, @ ...) or it is Enter.
function isCharacterKey(event) {
  if (event.ctrlKey || event.metaKey || event.altKey) return false; // shortcuts
  return event.key.length === 1 || event.key === "Enter";
}

// STOP typing at the limit with preventDefault()
textBox.addEventListener("keydown", event => {
  const hasSelection = textBox.selectionStart !== textBox.selectionEnd; // typing replaces the selection
  if (textBox.value.length >= MAX_CHARACTERS && isCharacterKey(event) && !hasSelection) {
    event.preventDefault();
  }
});

// Pasting could also exceed the limit, so only the part that fits is inserted
textBox.addEventListener("paste", event => {
  event.preventDefault();
  const pastedText = event.clipboardData.getData("text");
  const selectedLength = textBox.selectionEnd - textBox.selectionStart;
  const freeSpace = MAX_CHARACTERS - (textBox.value.length - selectedLength);
  if (freeSpace <= 0) return;

  textBox.setRangeText(pastedText.slice(0, freeSpace), textBox.selectionStart, textBox.selectionEnd, "end");
  updateCounter();
});

// Safety net for other input methods (drag and drop, mobile keyboards)
textBox.addEventListener("input", () => {
  if (textBox.value.length > MAX_CHARACTERS) {
    textBox.value = textBox.value.slice(0, MAX_CHARACTERS);
  }
  updateCounter();
});

// Reset clears everything
resetButton.addEventListener("click", () => {
  textBox.value = "";
  updateCounter();
  textBox.focus();
});

updateCounter();

HOW TO TEST IT (open q2.html in the browser)
---------------------------------------------
1. Type text: the counter goes down from 100 characters left.
2. At 20 characters left the counter turns YELLOW.
3. At 0 it turns RED. Keep typing: nothing is added (also Enter is blocked).
4. Press Backspace: it works, the counter goes back to yellow (1 left).
5. Paste a very long text: only the first 100 characters are inserted.
6. Click Reset: the box is empty, the counter shows 100 and has no colour.

EXPLANATION
-----------
event.preventDefault() on keydown cancels the browser's default action, which
is "insert the character". Letters are recognised with event.key.length === 1.
Yellow text on a white background is hard to read, so the counter is a
coloured badge with dark text on yellow and white text on red.

-------
Create a 3-step form: Step 1: Name, Step 2: Email, Step 3: Password.
"Next" moves to next step only if input is valid. "Back" moves to previous.

  validator returns an error, the error is shown and the step does not change.
* Back goes to the previous step; the typed values stay in the inputs.
* A progress bar shows the steps (current = blue, completed = green).
* After step 3 a summary is shown. The password is never displayed, only as
  dots, and "Start again" resets everything.

CODE - FILE 2 of 2:  q3.js  (the JavaScript of this question)
------------------------------------------------------------
"use strict";

const form = document.getElementById("multiStepForm");
const steps = Array.from(document.querySelectorAll(".step"));
const progressSteps = Array.from(document.querySelectorAll(".progress-step"));
const backButton = document.getElementById("backButton");
const nextButton = document.getElementById("nextButton");
const summaryBox = document.getElementById("summary");

const nameInput = document.getElementById("nameInput");
const emailInput = document.getElementById("emailInput");
const passwordInput = document.getElementById("passwordInput");

let currentStep = 0; // index of the visible step (0, 1 or 2)

// ---------- Validators: return an error message, or "" when the value is valid ----------
const validators = [
  () => {
    const name = nameInput.value.trim();
    if (name === "") return "Name is required.";
    if (!/^[A-Za-z][A-Za-z .'-]{1,49}$/.test(name)) return "Enter a valid name (letters only, at least 2 characters).";
    return "";
  },
  () => {
    const email = emailInput.value.trim();
    if (email === "") return "Email is required.";
    if (!/^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$/.test(email)) return "Enter a valid email (name@example.com).";
    return "";
  },
  () => {
    const password = passwordInput.value;
    if (password === "") return "Password is required.";
    if (password.length < 6) return "Password must be at least 6 characters.";
    if (!/[A-Za-z]/.test(password) || !/\d/.test(password)) return "Password must contain a letter and a number.";
    return "";
  }
];

// Shows or clears the inline error of one step
function showError(stepIndex, message) {
  const input = steps[stepIndex].querySelector("input");
  steps[stepIndex].querySelector(".error").textContent = message;
  input.classList.toggle("invalid", message !== "");
}

// Shows only the current step and updates the buttons and the progress bar
function showStep(stepIndex) {
  currentStep = stepIndex;
  steps.forEach((step, index) => step.classList.toggle("active", index === currentStep));
  progressSteps.forEach((item, index) => {
    item.classList.toggle("active", index === currentStep);
    item.classList.toggle("done", index < currentStep);
  });
  backButton.hidden = currentStep === 0;
  nextButton.textContent = currentStep === steps.length - 1 ? "Finish" : "Next";
  steps[currentStep].querySelector("input").focus();
}

function showSummary() {
  document.getElementById("summaryName").textContent = nameInput.value.trim();
  document.getElementById("summaryEmail").textContent = emailInput.value.trim();
  // The real password is never displayed, only its length
  document.getElementById("summaryPassword").textContent = "•".repeat(passwordInput.value.length);
  form.hidden = true;
  document.getElementById("progress").hidden = true;
  summaryBox.hidden = false;
}

// "Next" is the submit button, so pressing Enter also works.
form.addEventListener("submit", event => {
  event.preventDefault();

  const errorMessage = validators[currentStep]();
  showError(currentStep, errorMessage);
  if (errorMessage !== "") return; // stay on this step until the input is valid

  if (currentStep < steps.length - 1) {
    showStep(currentStep + 1);
  } else {
    showSummary(); // all 3 steps are complete
  }
});

"use strict";

const themeButtons = document.querySelectorAll("[data-theme-choice]");
const currentThemeLabel = document.getElementById("currentTheme");
const currentClassLabel = document.getElementById("currentClass");
const STORAGE_KEY = "selectedTheme";

// Applies a theme. EVERYTHING is done with setAttribute():
//   1. the class of <body>        -> body.theme-dark  (CSS reads this class)
//   2. the custom attribute       -> data-theme="dark" (saves the current theme)
//   3. the pressed state of the buttons (aria-pressed)
function applyTheme(themeName) {
  document.body.setAttribute("class", `theme-${themeName}`);
  document.body.setAttribute("data-theme", themeName);

  themeButtons.forEach(button => {
    const isSelected = button.getAttribute("data-theme-choice") === themeName;
    button.setAttribute("aria-pressed", String(isSelected));
  });

  // The labels read the values BACK from the attributes
  currentThemeLabel.textContent = document.body.getAttribute("data-theme");
  currentClassLabel.textContent = document.body.getAttribute("class");
}

themeButtons.forEach(button => {
  button.addEventListener("click", () => {
    const themeName = button.getAttribute("data-theme-choice");
    applyTheme(themeName);
    try {
      localStorage.setItem(STORAGE_KEY, themeName); // remember the choice after refresh
    } catch (error) {
      // storage can be blocked (private mode); the theme still works
    }
  });
});

// Start with the saved theme (if any), otherwise Light
let startTheme = "light";
try {
  const savedTheme = localStorage.getItem(STORAGE_KEY);
  if (["light", "dark", "blue"].includes(savedTheme)) startTheme = savedTheme;
} catch (error) {
  // ignore
}
applyTheme(startTheme);

"use strict";

// 6 images. Thumbnails use a small size, the modal asks for a BIGGER version
// of the same photo. Replace the ids / URLs with your own images if you like.
const photoIds = [10, 20, 30, 40, 50, 60];
const images = photoIds.map((id, index) => ({
  caption: `Photo ${index + 1}`,
  thumbnail: `https://picsum.photos/id/${id}/400/300`,
  large: `https://picsum.photos/id/${id}/1000/700`
}));

const gallery = document.getElementById("gallery");
const modalOverlay = document.getElementById("modalOverlay");
const modalContent = document.getElementById("modalContent");
const modalImage = document.getElementById("modalImage");
const modalCaption = document.getElementById("modalCaption");
const closeButton = document.getElementById("closeButton");

// If the internet image cannot load (offline), a coloured placeholder is shown
// instead, so the gallery always works.
function placeholderImage(label, width, height) {
  const svg =
    `<svg xmlns="http://www.w3.org/2000/svg" width="${width}" height="${height}">` +
    `<rect width="100%" height="100%" fill="#5b7db1"/>` +
    `<text x="50%" y="50%" fill="#fff" font-size="${Math.round(height / 7)}" font-family="Arial" ` +
    `text-anchor="middle" dominant-baseline="middle">${label}</text></svg>`;
  return `data:image/svg+xml;charset=utf-8,${encodeURIComponent(svg)}`;
}

function setImageSource(imageElement, url, label, width, height) {
  imageElement.onerror = () => {
    imageElement.onerror = null; // avoid an endless error loop
    imageElement.src = placeholderImage(label, width, height);
  };
  imageElement.src = url;
}

// ---------- Build the grid of 6 images ----------
images.forEach((imageData, index) => {
  const thumbnail = document.createElement("img");
  thumbnail.alt = imageData.caption;
  thumbnail.dataset.index = index;
  setImageSource(thumbnail, imageData.thumbnail, imageData.caption, 400, 300);
  gallery.appendChild(thumbnail);
});

function openModal(index) {
  const imageData = images[index];
  modalImage.alt = imageData.caption;
  setImageSource(modalImage, imageData.large, imageData.caption, 1000, 700);
  modalCaption.textContent = imageData.caption;
  modalOverlay.hidden = false;
}

function closeModal() {
  modalOverlay.hidden = true;
  modalImage.removeAttribute("src");
}

// Event delegation: one listener on the grid handles clicks on all images
gallery.addEventListener("click", event => {
  const clickedImage = event.target.closest("img");
  if (clickedImage) openModal(Number(clickedImage.dataset.index));
});

// Click on the dark area (outside the modal box) closes the modal
modalOverlay.addEventListener("click", closeModal);

// Click INSIDE the modal box: stopPropagation() stops the event from
// bubbling up to modalOverlay, so the overlay's close handler never runs.
modalContent.addEventListener("click", event => {
  event.stopPropagation();
});

// The close button is inside the box, so it closes the modal itself
closeButton.addEventListener("click", closeModal);

// Extra: the Esc key closes the modal
document.addEventListener("keydown", event => {
  if (event.key === "Escape" && !modalOverlay.hidden) closeModal();
});

-----------
Events bubble from the clicked element up through its parents. Without
stopPropagation(), the click on the picture inside the modal would bubble to

"use strict";

const trackingBox = document.getElementById("trackingBox");
const clientCoords = document.getElementById("clientCoords");
const boxCoords = document.getElementById("boxCoords");
const dotCount = document.getElementById("dotCount");
const clearButton = document.getElementById("clearButton");

// Converts a mouse event to a position INSIDE the box.
// clientX / clientY are measured from the top-left corner of the WINDOW, but a
// dot must be placed relative to the BOX, so the box's own position (and its
// border) is subtracted.
function getPositionInsideBox(event) {
  const boxRectangle = trackingBox.getBoundingClientRect();
  return {
    x: Math.round(event.clientX - boxRectangle.left - trackingBox.clientLeft),
    y: Math.round(event.clientY - boxRectangle.top - trackingBox.clientTop)
  };
}

// Live coordinates while the mouse moves inside the box
trackingBox.addEventListener("mousemove", event => {
  const position = getPositionInsideBox(event);
  clientCoords.textContent = `clientX: ${event.clientX}, clientY: ${event.clientY}`;
  boxCoords.textContent = `x: ${position.x}, y: ${position.y}`;
});

trackingBox.addEventListener("mouseleave", () => {
  clientCoords.textContent = "mouse is outside the box";
  boxCoords.textContent = "-";
});

// Double-click inside the box drops a red dot exactly at the click position
trackingBox.addEventListener("dblclick", event => {
  const position = getPositionInsideBox(event);

  const dot = document.createElement("div");
  dot.className = "dot";
  dot.style.left = `${position.x}px`; // the CSS translate(-50%, -50%) centres the dot on the point
  dot.style.top = `${position.y}px`;
  trackingBox.appendChild(dot);

  dotCount.textContent = trackingBox.querySelectorAll(".dot").length;
});

clearButton.addEventListener("click", () => {
  trackingBox.querySelectorAll(".dot").forEach(dot => dot.remove());
  dotCount.textContent = "0";
});

pointer-events: none on the dot lets later double-clicks pass through the dot

"use strict";

const dropdown = document.getElementById("dropdown");
const dropdownButton = document.getElementById("dropdownButton");
const optionsList = document.getElementById("optionsList");
const selectedValue = document.getElementById("selectedValue");

function openDropdown() {
  optionsList.hidden = false;
  dropdownButton.setAttribute("aria-expanded", "true");
}

function closeDropdown() {
  optionsList.hidden = true;
  dropdownButton.setAttribute("aria-expanded", "false");
}

// The button shows or hides the options
dropdownButton.addEventListener("click", () => {
  if (optionsList.hidden) {
    openDropdown();
  } else {
    closeDropdown();
  }
});

// Clicking an option updates the button text and closes the list
optionsList.addEventListener("click", event => {
  const option = event.target.closest("li");
  if (!option) return;

  optionsList.querySelectorAll("li").forEach(item => item.removeAttribute("aria-selected"));
  option.setAttribute("aria-selected", "true");

  dropdownButton.textContent = option.textContent;
  selectedValue.textContent = `You selected: ${option.textContent}`;
  closeDropdown();
});

// ---------- CLOSING WITH THE CAPTURING PHASE ----------
// The third argument { capture: true } registers the listener for the CAPTURING
// phase. A click travels window -> document -> ... -> target (capturing) and
// then back up (bubbling). A capturing listener on the document therefore runs
// FIRST, before any other handler on the page. Even if some other element calls
// event.stopPropagation() (which would stop a normal bubbling listener from
// ever seeing the click), the dropdown still closes.
document.addEventListener("click", event => {
  const clickedInsideDropdown = dropdown.contains(event.target);
  if (!clickedInsideDropdown && !optionsList.hidden) {
    closeDropdown();
  }
}, { capture: true });

// Extra: the Esc key closes the dropdown
document.addEventListener("keydown", event => {
  if (event.key === "Escape") closeDropdown();
});

3. Open it and click on an empty part of the page: it closes.
4. Open it and click inside the dropdown: it does not close by mistake.

EXPLANATION
-----------
The third argument of addEventListener can be true or { capture: true }. In
the capturing phase the handler sees the click before the target does, which
makes the "click outside" rule reliable. The test run also proved this: with a
bubbling handler on <body> that calls stopPropagation(), the dropdown still
closed.

"use strict";

const form = document.getElementById("signupForm");
const successMessage = document.getElementById("successMessage");

// One rule per field: returns an error message, or "" when the value is valid
const fields = [
  {
    input: document.getElementById("nameField"),
    error: document.getElementById("nameError"),
    validate: value => (value.trim() === "" ? "Name is required." : "")
  },
  {
    input: document.getElementById("emailField"),
    error: document.getElementById("emailError"),
    validate: value => {
      if (value.trim() === "") return "Email is required.";
      // must contain "@" with some text before and after it
      if (!/^[^\s@]+@[^\s@]+$/.test(value.trim())) return "Email must contain @ (example: name@mail.com).";
      return "";
    }
  },
  {
    input: document.getElementById("passwordField"),
    error: document.getElementById("passwordError"),
    validate: value => (value.length < 6 ? "Password must be at least 6 characters." : "")
  }
];

// Checks one field and shows or removes its inline error. Returns true if valid.
function validateField(field) {
  const message = field.validate(field.input.value);
  field.error.textContent = message;
  field.input.classList.toggle("invalid", message !== "");
  return message === "";
}

// LIVE errors: while the user types, the field is checked again, so an error
// disappears by itself as soon as the value becomes correct. Errors only
// appear after the first submit attempt or after leaving the field once.
const touchedFields = new Set();

fields.forEach(field => {
  field.input.addEventListener("blur", () => {
    touchedFields.add(field);
    validateField(field);
  });

  field.input.addEventListener("input", () => {
    successMessage.hidden = true;           // the old success message is no longer true
    if (touchedFields.has(field)) validateField(field);
  });
});

form.addEventListener("submit", event => {
  event.preventDefault(); // STOP the real form submission (no page reload)

  fields.forEach(field => touchedFields.add(field));
  const results = fields.map(validateField); // map() checks EVERY field, no early exit
  const isFormValid = results.every(Boolean);

  if (isFormValid) {
    successMessage.hidden = false;
    form.reset();                  // clear the fields for the next entry
    touchedFields.clear();
    fields.forEach(field => {
      field.error.textContent = "";
      field.input.classList.remove("invalid");
    });
  } else {
    successMessage.hidden = true;
    const firstInvalid = fields.find(field => field.input.classList.contains("invalid"));
    if (firstInvalid) firstInvalid.input.focus();
  }
});

HOW TO TEST IT (open q9.html in the browser)
---------------------------------------------
1. Click Submit with empty fields: three red inline errors appear.
2. Type a name: the name error disappears at once.
3. Type "abc" as email: the error mentions @. Type "ansh@mail.com": gone.
4. Type "123" as password: error. Type "123456": gone.
5. Click Submit: "Form Submitted Successfully" appears, the form is cleared.
   The page does not reload (the address bar does not change).

EXPLANATION
-----------
Without preventDefault() the browser would send the form and reload the page
(a normal form submission), so the success message would never be seen. In a
real application, the code that sends the data to a server would be added
inside the valid branch.
